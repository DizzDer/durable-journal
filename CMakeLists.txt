cmake_minimum_required(VERSION 3.16)
project(durable_journal VERSION 1.0.0 LANGUAGES CXX)
add_library(durable_journal INTERFACE)
target_compile_features(durable_journal INTERFACE cxx_std_17)
target_include_directories(durable_journal INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(journal_cli examples/journal_cli.cpp)
target_link_libraries(journal_cli PRIVATE durable_journal)
include(CTest)
if(BUILD_TESTING)
  add_executable(journal_tests tests/journal_tests.cpp)
  target_link_libraries(journal_tests PRIVATE durable_journal)
  add_test(NAME journal_tests COMMAND journal_tests)
endif()
